The signal equation of MRI is given by
\begin{equation}
    S_c(t) = \int_V s_c(\mathbf{r})\rho(\mathbf{r}, t) e^{-i\gamma\int_0^t \Delta B(\mathbf{r},\tau) d\tau} d\mathbf{r}
\end{equation}
The difference $\Delta B$ from the main magnetic field $B_0$ can be expressed like
\begin{equation}
    \Delta B(\mathbf{r}, t) = 
        \Delta B_0(\mathbf{r}) + \hspace{-4mm}
        \sum_{\beta \in \{x,y,z\}} \hspace{-4mm} G^0_\beta(t)r_\beta  + \hspace{-4mm}
        \sum_{\substack{|l|,|m| \geq 0 \\ \beta \in \{x,y,z\}}} \hspace{-4mm} G^{lm}_\beta(t) N^m_l(\mathbf{r})
\end{equation}
Where $N^m_l(\mathbf{r})$ are scalar fields describing the non-linear spatial variations of the magnetic field,
typically described either by the solid harmonics
\begin{equation}
    R^m_l(\mathbf{r}) = \norm{\mathbf{r}}^l Y^m_l(\mathbf{r}), \text{ for spherical harmonics } Y^m_l(\mathbf{r})
\end{equation}
or by the 



In the ideal case, the higher order terms from the harmonics are zero, and only the linear terms $G^0_\beta(t)r_\beta$ remain.
However, in practice there are always some higher order terms present, and the ideal gradient fields precsribed are the actual 
gradient fields realized. The actual gradient waveforms can be modelled well by a convolution of the ideal gradient waveforms 
with an impulse response function $\alpha(t)$. This can be expressed for all orders of the gradient fields as
\begin{equation}
    G^{lm}_\beta(t) = (\alpha^{lm}_\beta * G^{\text{ideal}}_\beta)(t)
\end{equation}
Or if gradient heating is to be taken into account, the impulse response function can be made time-varying as
\begin{equation}
    G^{lm}_\beta(t) = (\alpha^{lm}_\beta(t) * G^{\text{ideal}}_\beta)(t).
\end{equation}
While the magnetization $\rho(\mathbf{r}, t)$ is time dependent, we consider readout to be short enough that the only
time dependence is due to $T_2$ decay. Thus, we can express the magnetization as
\begin{equation}
    S_c(t) = \int_V s_C(\mathbf{r})\rho(\mathbf{r}) e^{-t/T_2(\mathbf{r})} e^{-i\gamma\int_0^t \Delta B(\mathbf{r},\tau) d\tau} d\mathbf{r}
\end{equation}
With this we can gather all terms that makes the signal equation deviate from a Fourier transform of the
static magnetization in the linear gradient fields and form
\begin{equation}
    S_c(t) = \int_V s_C(\mathbf{r})\rho(\mathbf{r})e^{\Phi(\mathbf{r}, t)} e^{-i\mathbf{k}(t)\cdot\mathbf{r}} d\mathbf{r} \\
\end{equation}
where
\begin{align}
    \Phi(\mathbf{r}, t) &= 
        -t/T_2(\mathbf{r}) - i\gamma\int_0^t \Delta B_0(\mathbf{r})d\tau + 
        \sum_{\substack{|l|,|m| > 0 \\ \beta \in \{x,y,z\}}} N^m_l(\mathbf{r}) \int_0^t G^{lm}_\beta(\tau) d\tau
    \\
    \mathbf{k}(t) &= \gamma\int_0^t (G^0_x(\tau), G^0_y(\tau), G^0_z(\tau)) d\tau
\end{align}
Considering a voxel discretization, we can express the operators as matrices by
\begin{align*}
    (\mathbf{A}_c)_{kj} &= s_C(\mathbf{r}_j)e^{\Phi(\mathbf{r}_j, t_k)} e^{-i\mathbf{k}(t_k)\cdot\mathbf{r}_j} \\
    (\mathbf{A}^H_c)_{ik} &= s^*_C(\mathbf{r}_i)e^{\Phi^*(\mathbf{r}_i, t_k)} e^{i\mathbf{k}(t_k)\cdot\mathbf{r}_i} \\
    (\mathbf{A}^H_c\mathbf{A}_c)_{ij} &= \sum_k s_C(\mathbf{r}_j)s^*_C(\mathbf{r}_i)e^{\Phi(\mathbf{r}_j, t_k) + \Phi^*(\mathbf{r}_i, t_k)} e^{-i\mathbf{k}(t_k)\cdot(\mathbf{r}_j - \mathbf{r}_i)}
\end{align*}

\section{Approximations for fast application of the normal operator}
To get back a Fourier transform, and to enable the Toeplitz property in $\mathbf{A}^H_c\mathbf{A}_c$. We introduce 
approximations
\begin{align}
    \exp(\Phi(\mathbf{r}_j, t_k)) &\approx \sum_{l=0}^L \Omega_{kl} \Upsilon_{jl} \label{eq:approx1}
\end{align}
This enables us to approximate the forward and backward operators by evaluating $L$ fourier transforms. 
We wish to get a similar decomposition for the normal operator. However, a naive application of the above approximation would give us a 
decomposition with $L^2$ Fourier transforms, which is infeasible. We thus try to get a decomposition into fewer Fourier terms.
We begin with the naive multiplication
\begin{equation}
    \exp(\Phi(\mathbf{r}_j, t_k) + \Phi^*(\mathbf{r}_i, t_k)) 
        \approx \sum_{l_1=0}^L \sum_{l_2=0}^L (\Omega_{kl_1} \Omega^*_{kl_2})(\Upsilon_{jl_1} \Upsilon^*_{il_2}) \label{eq:normal_op}
\end{equation}

\subsection{Choosing decompositions}
\subsubsection{The exponential decomposition}

Define $A^k_{ij} = \exp(\Phi(\mathbf{r}_j, t_k) + \Phi^*(\mathbf{r}_i, t_k)) = g_{jk} g^*_{ik}$. Where we further
defined $g_{jk} := \exp(\Phi(\mathbf{r}_j, t_k))$. 
We seek a approximation of $A^k_{ij}$ of the form
\begin{equation}
    \hat{A}^k_{ij} = \sum_{p=0}^P \Lambda_{kp} \Theta_{jp} \Theta^*_{ip}
\end{equation}
We use the usual Frobenius norm and seek to minimize the loss
\begin{equation}
    \mathcal{L} = \sum_k \norm{A^k - \hat{A}^k}_F^2
\end{equation}
Using Wirtinger calculus, we can compute the gradient with respect to $\Theta_{jp}$ as
\begin{equation}
    \frac{\partial \mathcal{L}}{\partial \Theta_{jp}^*} = \sum_{k,i}\bigg(\sum_{p' = 0}^P \Lambda_{kp'} \Theta_{jp'} \Theta^*_{ip'} - A^k_{ij}\bigg)\Lambda_{kp}^* \Theta_{ip}
\end{equation}
and the gradient with respect to $\Lambda_{kp}$ as
\begin{equation}
    \frac{\partial \mathcal{L}}{\partial \Lambda_{kp}^*} = \sum_{j,i}\bigg(\sum_{p' = 0}^P \Lambda_{kp'} \Theta_{jp'} \Theta^*_{ip'} - A^k_{ij}\bigg)\Theta_{jp} \Theta^*_{ip}
\end{equation}
If we have orthogonality constraints on $\Theta_{jp}$, 
\begin{equation}
    \sum_j \Theta^*_{jp} \Theta_{jq} = \delta_{pq},
\end{equation}
then the gradients simplify.
\begin{equation}
\begin{split}
    \frac{\partial \mathcal{L}}{\partial \Theta_{jp}^*} 
    &= \sum_{k,i}\bigg(\sum_{p'=0}^P \Lambda_{kp'} \Theta_{jp'} \Theta^*_{ip'} - g_{jk} g^*_{ik}\bigg)\Lambda_{kp}^* \Theta_{ip} 
    \\
    &= \sum_k\sum_{p'=0}^P \Lambda_{kp'} \Lambda_{kp}^* \Theta_{jp'}\bigg(\sum_i \Theta^*_{ip'}\Theta_{ip}\bigg)
      - \sum_k g_{jk}\Lambda_{kp}^* \sum_i g^*_{ik} \Theta_{ip} 
    \\
    &= \sum_k \left[|\Lambda_{kp}|^2 \Theta_{jp} - g_{jk}\Lambda_{kp}^* \sum_i g^*_{ik} \Theta_{ip} \right]
\end{split}
\end{equation}
and
\begin{equation}
\begin{split}
    \frac{\partial \mathcal{L}}{\partial \Lambda_{kp}^*} 
    &= \sum_{j,i}\bigg(\sum_{p'=0}^P \Lambda_{kp'} \Theta_{jp'} \Theta^*_{ip'} - g_{jk} g^*_{ik}\bigg)\Theta_{jp}^* \Theta_{ip} 
    \\
    &= \sum_{p'=0}^P \Lambda_{kp'} \bigg(\sum_{j,i} \Theta_{jp'} \Theta^*_{ip'} \Theta_{jp}^* \Theta_{ip}\bigg) - \sum_{j,i} g_{jk} g^*_{ik} \Theta_{jp}^* \Theta_{ip} 
    \\
    &= \sum_{p'=0}^P \Lambda_{kp'} \bigg(\sum_j \Theta_{jp'} \Theta_{jp}^*\bigg)\bigg(\sum_i \Theta^*_{ip'} \Theta_{ip}\bigg) - \sum_{j,i} g_{jk} g^*_{ik} \Theta_{jp}^* \Theta_{ip} 
    \\
    &= \sum_{p'=0}^P \Lambda_{kp'} \delta_{pp'} - \sum_{j,i} g_{jk} g^*_{ik} \Theta_{jp}^* \Theta_{ip} 
    \\
    &= \Lambda_{kp} - (\sum_j g_{jk} \Theta_{jp}^*)(\sum_ig^*_{ik} \Theta_{ip}) 
    \\
    &= \Lambda_{kp} - |\sum_i g_{ik} \Theta_{ip}|^2
\end{split}
\end{equation}
Thus the optimal $\Lambda_{kp}$ for fixed $\Theta_{jp}$ is given by
\begin{equation}
    \Lambda_{kp} = |\sum_i g_{ik} \Theta_{ip}|^2.
\end{equation}

Thus we arrive at the following alternating optimization scheme:
\begin{enumerate}[label=\arabic*.]
    \item Initialize $\Theta_{jp}$. If not initialy orthogonal, orthogonalize.
    \item Update $\Lambda_{kp} = |\sum_i g_{ik} \Theta_{ip}|^2$.
    \item Update $\Theta_{jp} = \sum_k \Lambda_{kp} g_{jk} \sum_i g^*_{ik} \Theta_{ip}$, then orthogonalize.
    \item Repeat steps 2 and 3 until convergence.
\end{enumerate}




\subsubsection{SVD-optimal initialisation of $\Theta$}

The alternating scheme above requires an initialisation for $\Theta_{jp}$.
We derive the choice of $\Theta_{jp}$ that is optimal for minimising the
summed normal-operator error $\sum_{i,j}|N_{ij} - \hat{N}_{ij}|^2$, where
\begin{equation}
    N_{ij} = \sum_k g^*_{ik} g_{jk}, \qquad
    \hat{N}_{ij} = \sum_p \mu_p \Theta^*_{ip} \Theta_{jp},
    \qquad \mu_p = \sum_k \Lambda_{kp}.
\end{equation}
Note the conjugate sits on the $i$-index in both $N_{ij}$ and $\hat{N}_{ij}$,
consistent with the approximation $g_{jk} \approx \sum_p \Theta_{jp}$ (forward model
uses $\Theta_{jp}$ plain, not conjugated).

Taking the element-wise conjugate of both sides (which does not change $\sum_{i,j}|\cdot|^2$):
\begin{equation}
    \sum_{i,j}|N_{ij} - \hat{N}_{ij}|^2
    = \sum_{i,j}\left|\sum_k g_{ik}g^*_{jk} - \sum_p \mu_p \Theta_{ip}\Theta^*_{jp}\right|^2.
    \label{eq:svd_equiv_loss}
\end{equation}
The right-hand side is the Frobenius loss for approximating the matrix
$\tilde{N}_{ij} := \sum_k g_{ik}g^*_{jk}$
by the rank-$P$ Hermitian form $\sum_p \mu_p \Theta_{ip}\Theta^*_{jp}$.
By the Eckart--Young theorem, the minimiser over orthonormal $\{\Theta_{jp}\}$ is
the $P$ leading eigenvectors of $\tilde{N}_{ij}$.

Now observe that $\tilde{N}_{ij} = \sum_k g_{ik}g^*_{jk}$ is exactly the normal operator
of the \emph{conjugated} forward model $g^*_{jk}$.
The forward approximation \eqref{eq:approx1} gives
\begin{equation}
    g_{jk} \approx \sum_l \Omega_{kl}\Upsilon_{jl}.
\end{equation}
Conjugating: $g^*_{jk} \approx \sum_l \Omega^*_{kl}\Upsilon^*_{jl}$.
The normal operator of this conjugated model is
\begin{equation}
    \tilde{N}_{ij} = \sum_k g_{ik}g^*_{jk}
    \approx \sum_l |\sigma_l|^2\, \Upsilon^*_{il}\Upsilon_{jl},
\end{equation}
where $\sigma_l$ are the singular values coming from the $\Omega_{kl}$ orthogonality.
The eigenvectors of $\tilde{N}_{ij}$ are therefore $\Upsilon_{jp}$,
the same spatial modes already appearing in the decomposition of $g_{jk}$.

Thus the \textbf{optimal initialisation} is
\begin{equation}
    \Theta_{jp} = \Upsilon_{jp}, \qquad \mu_p = \sigma_p^2.
    \label{eq:svd_init}
\end{equation}
Substituting into the closed-form optimal $\Lambda_{kp}$:
\begin{equation}
\begin{split}
    \Lambda_{kp} &= \left|\sum_i g^*_{ik}\,\Theta_{ip}\right|^2
    = \left|\sum_i g^*_{ik}\,\Upsilon_{ip}\right|^2 \\
    &= \left|\sum_i \sum_l \Omega^*_{kl}\Upsilon^*_{il}\,\Upsilon_{ip}\right|^2
    = \left|\Omega^*_{kp}\right|^2 = |\Omega_{kp}|^2,
\end{split}
\end{equation}
where the last step uses the orthonormality $\sum_i\Upsilon^*_{il}\Upsilon_{ip}=\delta_{lp}$.
Thus
\begin{equation}
    \Lambda_{kp} = |\Omega_{kp}|^2.
    \label{eq:lambda_svd}
\end{equation}
The per-$k$ weight $\Lambda_{kp}$ is the squared magnitude of the $k$-space coefficient
for mode $p$, which determines the Toeplitz kernel
$h_p = \mathbf{F}_{\mathrm{adj}}(\Lambda_{\cdot p})$ for the fast normal-operator application.

In practice $g_{jk}$ is not available in factored form; the optimal $\Theta_{jp}$
is obtained via a randomised eigendecomposition of $\tilde{N}_{ij} = \sum_k g_{ik}g^*_{jk}$
(or equivalently of $N_{ij}$, whose conjugate eigenvectors equal those of $\tilde{N}$),
keeping the $P$ eigenvectors with largest eigenvalues.

\subsubsection{Decomposition of $\Omega_{kl_1} \Omega^*_{kl_2}$}

For each $k$ we have a $L\times L$ hermitian matrix, thus we can extract the $L(L+1)/2$ unique elements to form a $(L(L+1)/2) \times K$ matrix on which
we can perform a SVD truncated to the $P$'th largest singular values and reform the $L \times L$ matrix to get
\begin{equation}
    \Omega_{kl_1} \Omega^*_{kl_2} \approx \sum_{p=0}^P C_{l_1l_2,p} \Lambda_{kp}. \label{eq:time_approx_SVD}
\end{equation}
For each $p$ we have a $L \times L$ hermitian matrix, for which we can perform a eigen-decomposition 
into the rank $R$ approximation as
\begin{equation}
    C_{l_1l_2,p} \approx \sum_{r=0}^R \lambda_{rp} \hat{C}_{l_1rp} \hat{C}^*_{l_2rp}.
\end{equation}
With $R = 1$ in \eqref{eq:time_approx_SVD} this would correspond to a Canonical Polyadic Decomposition.

Inserting these approximations into the normal operator \eqref{eq:normal_op} we get
\begin{align}
    \exp(\Phi(\mathbf{r}_j, t_k) + \Phi^*(\mathbf{r}_i, t_k)) 
        &\approx \sum_{l_1=0}^L \sum_{l_2=0}^L \sum_{p=0}^P \sum_{r=0}^R \Lambda_{kp} \lambda_{rp} \hat{C}_{l_1rp} \hat{C}^*_{l_2rp} \Upsilon_{jl_1} \Upsilon^*_{il_2} \\
        &= \sum_{p=0}^P \sum_{r=0}^R \Lambda_{kp} \lambda_{rp} \left(\sum_{l_1=0}^L \hat{C}_{l_1rp} \Upsilon_{jl_1}\right) \left(\sum_{l_2=0}^L \hat{C}_{l_2rp} \Upsilon_{il_2}\right)^* \\
        &= \sum_{p=0}^P \sum_{r=0}^R \Lambda_{kp} \lambda_{rp} \Theta_{jrp} \Theta^*_{irp} \label{eq:final_approx_normal_op}
\end{align}

\subsubsection{Decomposition of $\Upsilon_{jl_1}\Upsilon^*_{il_2}$}

Instead of decomposing the temporal factors 
$\Omega_{kl_1}\Omega^*_{kl_2}$, we may instead seek a decomposition
of the spatial factors
\begin{equation}
    \Upsilon_{jl_1}\Upsilon^*_{il_2}.
\end{equation}
The goal is to obtain an approximation of the form
\begin{equation}
    \Upsilon_{jl_1}\Upsilon^*_{il_2}
    \approx
    \sum_{p=0}^{P}
    C_{l_1l_2,p}\Theta_{jp}\Theta^*_{ip},
    \label{eq:upsilon_target}
\end{equation}
which would reduce the number of Fourier transforms required for
application of the normal operator from $L^2$ to $P$.

Direct optimization of \eqref{eq:upsilon_target} is infeasible for realistic
problem sizes, since the tensors
$\Upsilon_{jl_1}\Upsilon^*_{il_2}$ cannot be explicitly formed or stored.
Instead, we construct the decomposition using only matrix-vector products.

We begin by viewing $\Upsilon_{jl}$ as a matrix
\begin{equation}
    \mathbf{\Upsilon}\in\mathbb{C}^{N\times L},
\end{equation}
where $N$ is the number of spatial voxels.
We then form the Hermitian operator
\begin{equation}
    S_{ji}
    =
    \sum_{l=0}^{L}
    \Upsilon_{jl}\Upsilon^*_{il}.
    \label{eq:upsilon_covariance}
\end{equation}
This operator is never explicitly assembled. Instead, products
$\mathbf{y}=\mathbf{S}\mathbf{x}$ are evaluated as
\begin{align}
    z_l
    &=
    \sum_i
    \Upsilon^*_{il}x_i,
    \\
    y_j
    &=
    \sum_l
    \Upsilon_{jl}z_l.
\end{align}
Thus, application of $\mathbf{S}$ requires only multiplication by
$\mathbf{\Upsilon}$ and $\mathbf{\Upsilon}^H$.

Using a Krylov subspace eigensolver such as Lanczos or randomized SVD,
we compute the leading $P$ eigenvectors of $\mathbf{S}$,
\begin{equation}
    \sum_i
    S_{ji}\Theta_{ip}
    =
    \sigma_p^2\Theta_{jp},
\end{equation}
with orthonormality
\begin{equation}
    \sum_j
    \Theta^*_{jp}\Theta_{jq}
    =
    \delta_{pq}.
\end{equation}
The vectors $\Theta_{jp}$ span the optimal rank-$P$ subspace in the
sense of the Eckart--Young theorem applied to $\mathbf{\Upsilon}$.

Projecting $\Upsilon_{jl}$ onto this basis gives
\begin{equation}
    B_{lp}
    =
    \sum_j
    \Theta^*_{jp}\Upsilon_{jl},
    \label{eq:B_projection}
\end{equation}
such that
\begin{equation}
    \Upsilon_{jl}
    \approx
    \sum_{p=0}^{P}
    \Theta_{jp}B_{lp}.
\end{equation}
Substituting into
$\Upsilon_{jl_1}\Upsilon^*_{il_2}$ yields
\begin{align}
    \Upsilon_{jl_1}\Upsilon^*_{il_2}
    &\approx
    \sum_{p=0}^{P}\sum_{q=0}^{P}
    \Theta_{jp}
    B_{l_1p}
    B^*_{l_2q}
    \Theta^*_{iq}.
\end{align}
The desired decomposition \eqref{eq:upsilon_target} is obtained by
neglecting off-diagonal couplings $p\neq q$, giving
\begin{equation}
    \Upsilon_{jl_1}\Upsilon^*_{il_2}
    \approx
    \sum_{p=0}^{P}
    B_{l_1p}B^*_{l_2p}
    \Theta_{jp}\Theta^*_{ip}.
\end{equation}
We therefore identify
\begin{equation}
    C_{l_1l_2,p}
    =
    B_{l_1p}B^*_{l_2p}.
\end{equation}

The quality of the approximation depends on the magnitude of the
discarded cross terms
$B_{l_1p}B^*_{l_2q}$ for $p\neq q$.
To reduce these couplings, we may further rotate the basis vectors
within the retained subspace.
Let
\begin{equation}
    \hat{\Theta}_{jp}
    =
    \sum_{q=0}^{P}
    \Theta_{jq}U_{qp},
\end{equation}
where $\mathbf{U}\in\mathbb{C}^{P\times P}$ is unitary.
The projected coefficients become
\begin{equation}
    \hat{B}_{lp}
    =
    \sum_q
    B_{lq}U_{qp}.
\end{equation}
The rotation matrix $\mathbf{U}$ may then be chosen to minimize the
off-diagonal energy
\begin{equation}
    \mathcal{E}(\mathbf{U})
    =
    \sum_l
    \sum_{p\neq q}
    |\hat{B}_{lp}\hat{B}^*_{lq}|^2.
\end{equation}
Since this optimization is performed only over the small
$P\times P$ unitary matrix $\mathbf{U}$, it remains computationally
feasible even for large imaging problems.
The optimization may be performed using Jacobi rotations,
Riemannian gradient descent on the unitary group,
or approximate joint diagonalization algorithms.



\subsection{How to compute interpolants efficiently}
Now, to find the approximations, performing the randomized SVD and SVD in these steps is computationally expensive, therefore, one can 
approach this with a histogram based method. First of all, the off-resonance and off-linear gradient terms only have effects on the signal on the support
of the image. Thus, we do these approximations first only on voxels that are within the support. Next we can consider
the feature vector
\begin{equation}
    \overline{\Psi}(\mathbf{r}) = \begin{bmatrix}
        \mathbf{z}(\mathbf{r}) = \Delta B_0(\mathbf{r}) + 1/ T_2(\mathbf{r}) \\
        R^0_0(\mathbf{r}) \\
        R^0_1(\mathbf{r}) \\
        R^{-1}_1(\mathbf{r}) \\
        \vdots
    \end{bmatrix}
\end{equation}
Now, we can make a histogram of the feature vectors for all voxels in the support, and perform the approximations on the histogram bins instead of on the individual voxels.

\section*{Normal Operator via Toeplitz Embedding}

Given the approximation \eqref{eq:approx1}, the discretised forward operator for coil $c$ is
\begin{equation}
    (\mathbf{A}_c \boldsymbol{\rho})_k
        = \sum_l \Omega_{kl}
          \bigl[\mathbf{F}\,\mathrm{diag}(\Upsilon_l \odot s_c)\,\boldsymbol{\rho}\bigr]_k,
\end{equation}
where $\mathbf{F}$ is the NUFFT and $(\Upsilon_l)_j = \Upsilon_{jl}$.
The adjoint is
\begin{equation}
    (\mathbf{A}_c^H \mathbf{s})_j
        = s_c^*(\mathbf{r}_j)
          \sum_l \Upsilon_{jl}^*\,\bigl[\mathbf{F}^H\,\mathrm{diag}(\Omega_l^*)\,\mathbf{s}\bigr]_j.
\end{equation}

\subsection*{Toeplitz structure}

Define the PSF for k-space weights $w \in \mathbb{C}^K$ as the type-1 NUFFT
\begin{equation}
    h_w = \mathbf{F}_{\mathrm{adj}}(w), \qquad
    (h_w)_n = \sum_k w_k\,e^{i\mathbf{k}(t_k)\cdot\mathbf{r}_n}.
    \label{eq:psf_def}
\end{equation}
The Toeplitz application uses a zero-padded FFT of size $2N$ in each dimension:
\begin{equation}
    \bigl[\mathbf{F}^H\,\mathrm{diag}(w)\,\mathbf{F}\,f\bigr]
        = h_w \star f
        = \mathrm{IFFT}\!\bigl[\hat{H}_w \odot \mathrm{FFT}(f_{\mathrm{pad}})\bigr],
    \label{eq:toeplitz_apply}
\end{equation}
where $\hat{H}_w = \mathrm{FFT}(h_{w,\mathrm{pad}})$ is precomputed once.

\subsection*{$L^2$ Toeplitz approach}

Using \eqref{eq:normal_op} directly:
\begin{equation}
    (\mathbf{A}_c^H\mathbf{A}_c\,\boldsymbol{\rho})_i
        \approx \sum_{l_1=1}^L \sum_{l_2=1}^L
            \Upsilon_{il_2}^*\;
            \bigl(h_{l_1 l_2} \star (\Upsilon_{l_1} \odot s_c \odot \boldsymbol{\rho})\bigr)_i,
    \label{eq:normal_op_Lsq}
\end{equation}
with one kernel per pair:
\begin{equation}
    h_{l_1 l_2} = \mathbf{F}_{\mathrm{adj}}(\Omega_{l_1} \odot \Omega_{l_2}^*).
\end{equation}
Setup cost: $L^2$ type-1 NUFFTs.
Application cost per iteration: $L^2$ Toeplitz multiplications \eqref{eq:toeplitz_apply}.

\subsection*{$P \times R$ Toeplitz approach (reduced)}

Inserting \eqref{eq:final_approx_normal_op}:
\begin{equation}
    (\mathbf{A}_c^H\mathbf{A}_c\,\boldsymbol{\rho})_i
        \approx \sum_{p=1}^P \sum_{r=1}^R
            \Theta_{irp}^*\;
            \bigl(h_p \star (\Theta_{rp} \odot \boldsymbol{\rho})\bigr)_i,
    \label{eq:normal_op_PR}
\end{equation}
where $P$ distinct kernels (one per $p$):
\begin{equation}
    h_p = \mathbf{F}_{\mathrm{adj}}(\Lambda_p),
\end{equation}
and the combined spatial weights:
\begin{equation}
    \Theta_{jrp} = s_c(\mathbf{r}_j)\sum_{l=1}^L \hat{C}_{lrp}\,\Upsilon_{jl}.
\end{equation}
Setup cost: $P$ type-1 NUFFTs (independent of $L$).
Application cost per iteration: $P \times R$ Toeplitz multiplications.
With $R=1$ this is $P$ multiplications, and if $P \ll L^2$ we achieve a large speedup.

\subsection*{Algorithm (setup)}
\begin{enumerate}[label=\arabic*.]
    \item Compute $\Omega \in \mathbb{C}^{K \times L}$, $\Upsilon \in \mathbb{C}^{N_{\mathrm{hist}} \times L}$ from SVD of $e^{\Phi}$.
    \item Form the $\frac{L(L+1)}{2} \times K$ matrix of unique upper-triangular products $\Omega_{kl_1}\Omega_{kl_2}^*$.
    \item Truncated SVD of rank $P$ $\Rightarrow$ $\boldsymbol{\Lambda} \in \mathbb{C}^{K \times P}$, coefficient matrices $\mathbf{C}_p \in \mathbb{C}^{L \times L}$.
    \item Rank-$R$ SVD of each $\mathbf{C}_p$ $\Rightarrow$ $\hat{C}_{lrp}$.
    \item $\Theta_{jrp} = \sum_l \hat{C}_{lrp}\,\Upsilon_{jl}$ scattered to full voxel grid.
    \item $P$ Toeplitz kernels: $\hat{H}_p = \mathrm{FFT}\!\bigl[\mathbf{F}_{\mathrm{adj}}(\Lambda_p)_{\mathrm{pad}}\bigr]$.
\end{enumerate}

\subsection*{Normal operator application}
\begin{equation}
    (\mathbf{A}^H\mathbf{A}\,\boldsymbol{\rho})_i
        = \sum_{p=1}^P \sum_{r=1}^R
            \Theta_{irp}^*\;
            \mathrm{IFFT}\!\bigl[\hat{H}_p \odot \mathrm{FFT}((\Theta_{rp} \odot \boldsymbol{\rho})_{\mathrm{pad}})\bigr]_i.
\end{equation}



\section{Non-Hermitian Decomposition and Multi-Coil Normal Operator}

\subsection{Non-Hermitian Approximation}

The hermitian decomposition of the normal-operator phase factor
\begin{equation}
    e^{\Phi(\mathbf{r}_j, t_k) + \Phi^*(\mathbf{r}_i, t_k)} \approx \sum_{p=0}^P \Lambda_{kp}\,\Theta_{jp}\Theta^*_{ip}
\end{equation}
constrains the same spatial basis $\Theta$ to appear on both the $j$- and $i$-index, and forces $\Lambda_{kp} \geq 0$.
We generalise by allowing independent spatial functions for each index:
\begin{equation}
    e^{\Phi(\mathbf{r}_j, t_k) + \Phi^*(\mathbf{r}_i, t_k)}
    \approx \sum_{p=0}^P \Lambda_{kp}\,A_{jp}\,B_{ip},
    \label{eq:nonherm_decomp}
\end{equation}
where $A_{jp}$, $B_{ip} \in \mathbb{C}$ are independent spatial basis functions and $\Lambda_{kp} \in \mathbb{C}$.
Here $B$ is defined so that the hermitian case corresponds to $B_{ip} = \Theta^*_{ip}$, $A_{jp} = \Theta_{jp}$,
$\Lambda_{kp} \in \mathbb{R}_{\geq 0}$.
The more general form \eqref{eq:nonherm_decomp} strictly enlarges the approximation class and can achieve smaller error for the same number of terms $P$.

\subsection{Full Normal Operator with Coil Maps and Coil Covariance}

We consider $C$ receive coils with sensitivity maps $s_c(\mathbf{r})$ and noise covariance matrix $\Psi \in \mathbb{C}^{C \times C}$ (positive definite Hermitian).
The per-coil forward operator element is
\begin{equation}
    (A_c)_{kj} = s_{cj}\,e^{\Phi(\mathbf{r}_j, t_k)}\,e^{-i\mathbf{k}(t_k)\cdot\mathbf{r}_j},
\end{equation}
and the maximum-likelihood (whitened) normal operator is
\begin{equation}
    N_{ij}
    = \sum_{c,c'} (\Psi^{-1})_{cc'} \sum_k (A_c^H)_{ik}(A_{c'})_{kj}
    = \sum_{c,c'} (\Psi^{-1})_{cc'}\,s^*_{ci}\,s_{c'j}
      \sum_k e^{\Phi^*(\mathbf{r}_i,t_k)+\Phi(\mathbf{r}_j,t_k)}
             e^{-i\mathbf{k}(t_k)\cdot(\mathbf{r}_j - \mathbf{r}_i)}.
\end{equation}
Define the coil correlation factor
\begin{equation}
    \Xi_{ij}
    = \mathbf{s}_i^H\,\Psi^{-1}\,\mathbf{s}_j
    = \sum_{c,c'} (\Psi^{-1})_{cc'}\,s^*_{ci}\,s_{c'j},
    \quad [\mathbf{s}_j]_c = s_{cj}.
\end{equation}
Inserting the non-hermitian approximation \eqref{eq:nonherm_decomp} and defining the Toeplitz kernels
\begin{equation}
    h_p(\mathbf{r}) = \sum_k \Lambda_{kp}\,e^{-i\mathbf{k}(t_k)\cdot\mathbf{r}}
    = \bigl[\mathbf{F}_{\mathrm{adj}}(\Lambda_{\cdot p})\bigr](\mathbf{r}),
    \label{eq:nonherm_kernel}
\end{equation}
the full normal operator element becomes
\begin{equation}
    N_{ij}
    \approx
    \bigl(\mathbf{s}_i^H\,\Psi^{-1}\,\mathbf{s}_j\bigr)
    \sum_p A_{jp}\,B_{ip}\;h_p(\mathbf{r}_j - \mathbf{r}_i).
    \label{eq:nonherm_N}
\end{equation}
The Toeplitz convolution structure is preserved because $h_p$ depends only on the difference $\mathbf{r}_j - \mathbf{r}_i$.
Note that the kernels $h_p$ are complex-valued when $\Lambda_{kp} \in \mathbb{C}$.

\subsection{Operator Application and Whitened Coil Maps}

The action of \eqref{eq:nonherm_N} on a vector $\boldsymbol{\rho}$ is
\begin{equation}
    (N\boldsymbol{\rho})_i
    \approx
    \sum_p B_{ip}
    \sum_{c,c'} (\Psi^{-1})_{cc'}\,s^*_{ci}
    \Bigl[h_p \star \bigl(s_{c'} \odot A_p \odot \boldsymbol{\rho}\bigr)\Bigr]_i.
    \label{eq:nonherm_apply}
\end{equation}
To expose the structure more cleanly, factor $\Psi^{-1} = L^H L$ (e.g.\ via Cholesky) and define the
\emph{whitened coil maps}
\begin{equation}
    \tilde{s}_{qj} = \bigl[L\,\mathbf{s}_j\bigr]_q, \qquad q = 1,\dots,C.
\end{equation}
Then $\Xi_{ij} = \sum_q \tilde{s}^*_{qi}\,\tilde{s}_{qj}$ and \eqref{eq:nonherm_apply} becomes
\begin{equation}
    \boxed{
    (N\boldsymbol{\rho})_i
    \approx
    \sum_p B_{ip}
    \sum_q \tilde{s}^*_{qi}
    \Bigl[h_p \star \bigl(\tilde{s}_q \odot A_p \odot \boldsymbol{\rho}\bigr)\Bigr]_i.
    }
    \label{eq:nonherm_apply_whitened}
\end{equation}
The coil covariance enters only through the whitened maps $\tilde{s}_q$.
The per-iteration cost is $P \times C$ Toeplitz multiplications, identical in structure to the hermitian
case but with the flexibility of using different input ($A_p$) and output ($B_p$) spatial modes.
If $\Psi^{-1}$ is well-approximated by a rank-$Q$ matrix ($Q < C$), the inner sum
truncates to $Q$ virtual coils.

\subsubsection*{Per-mode algorithm}

For each mode $p = 1,\dots,P$:
\begin{enumerate}[label=\arabic*.]
    \item For each whitened coil $q$: form $v^{(q)}_j = \tilde{s}_{qj}\,A_{jp}\,\rho_j$.
    \item For each $q$: Toeplitz-convolve $u^{(q)}_i = \bigl[h_p \star v^{(q)}\bigr]_i$.
    \item Accumulate: $(N\boldsymbol{\rho})_i \mathrel{+}= B_{ip}\sum_q \tilde{s}^*_{qi}\,u^{(q)}_i$.
\end{enumerate}

\subsection{Optimal Decomposition: Alternating Optimization}

\subsubsection*{Scale and implicit operators}

In practice $N, K \sim 10^5$--$10^6$, so the matrix $G \in \mathbb{C}^{N \times K}$
with $G_{jk} = g_{jk} = e^{\Phi(\mathbf{r}_j,t_k)}\,e^{-i\mathbf{k}(t_k)\cdot\mathbf{r}_j}$
is never formed or stored.
All computations below involve only matrix-vector products of the form
\begin{equation}
    [G\,\mathbf{v}]_j = \sum_k g_{jk}\,v_k,
    \qquad
    [G^T\mathbf{x}]_k = \sum_j g_{jk}\,x_j,
    \qquad
    [\bar{G}\,\mathbf{v}]_j = \sum_k g^*_{jk}\,v_k,
    \qquad
    [\bar{G}^T\mathbf{x}]_k = \sum_j g^*_{jk}\,x_j,
\end{equation}
each of which is one (possibly conjugated) NUFFT application of cost $O(N\log N + K)$.
The SVD needed for initialisation is computed on a small histogram-reduced matrix
$G_{\mathrm{hist}} \in \mathbb{C}^{N_{\mathrm{hist}} \times K}$ with
$N_{\mathrm{hist}} \ll N$ as described in Section~\ref{sec:histogram}.

\subsubsection*{Loss and constraints}

We seek $A \in \mathbb{C}^{N \times P}$, $B \in \mathbb{C}^{N \times P}$,
$\Lambda \in \mathbb{C}^{K \times P}$ minimising
\begin{equation}
    \mathcal{L}
    = \sum_{i,j,k}
      \Bigl|g_{jk}g^*_{ik} - \sum_{p=0}^P \Lambda_{kp}A_{jp}B_{ip}\Bigr|^2,
    \label{eq:nonherm_loss}
\end{equation}
under the orthonormality constraints $A^H A = I_P$ and $B^H B = I_P$.

\subsubsection*{Update for $\Lambda$ (fixed $A$, $B$)}

Define the projected temporal matrices
\begin{equation}
    \mathbf{F} = G^T\overline{A} \in \mathbb{C}^{K \times P},
    \quad F_{kp} = \sum_j g_{jk}\,\overline{A_{jp}},
    \qquad
    \mathbf{C} = \overline{G}^T\overline{B} \in \mathbb{C}^{K \times P},
    \quad C_{kp} = \sum_j g^*_{jk}\,\overline{B_{jp}}.
    \label{eq:FC_def}
\end{equation}
Each column of $\mathbf{F}$ and $\mathbf{C}$ costs one NUFFT.
Setting $\partial\mathcal{L}/\partial\Lambda^*_{kp} = 0$ under $A^HA=I$, $B^HB=I$ gives
\begin{equation}
    \boldsymbol{\Lambda} = \mathbf{F} \odot \mathbf{C}.
    \label{eq:lambda_update}
\end{equation}
In the hermitian case ($B_{ip}=\Theta^*_{ip}$, $A_{jp}=\Theta_{jp}$) this reduces to
$\Lambda_{kp} = |\sum_j g_{jk}\overline{\Theta_{jp}}|^2 \geq 0$.
In general $\Lambda_{kp} \in \mathbb{C}$.

\subsubsection*{Update for $A$ (fixed $B$, $\Lambda$)}

Setting $\partial\mathcal{L}/\partial\overline{A_{jp}} = 0$ under $B^H B = I$ gives
\begin{equation}
    A_{\cdot p} \leftarrow G\bigl(\overline{\Lambda_{\cdot p}} \odot \mathbf{C}_{\cdot p}\bigr),
    \label{eq:A_update}
\end{equation}
i.e.\ one $G$-matvec per mode, followed by orthogonalisation of $A$.

\subsubsection*{Update for $B$ (fixed $A$, $\Lambda$)}

Setting $\partial\mathcal{L}/\partial\overline{B_{ip}} = 0$ under $A^H A = I$ gives
\begin{equation}
    B_{\cdot p} \leftarrow \bar{G}\bigl(\overline{\Lambda_{\cdot p}} \odot \mathbf{F}_{\cdot p}\bigr),
    \label{eq:B_update}
\end{equation}
i.e.\ one $\bar{G}$-matvec per mode (conjugate NUFFT), followed by orthogonalisation of $B$.
Note the symmetry: both $A$ and $B$ updates use $\overline{\Lambda_p}$, but $A$ uses $G$ and $B$ uses $\bar{G}$.

\subsubsection*{Cost per ALS iteration}

Each pass through updates \eqref{eq:lambda_update}--\eqref{eq:B_update} requires $4P$ NUFFT
applications (2 for $\mathbf{F},\mathbf{C}$; $P$ each for the $A$ and $B$ updates)
plus two $O(NP^2)$ Gram--Schmidt orthogonalisations, cheap for small $P$.

\subsubsection*{Algorithm}

\begin{enumerate}[label=\arabic*.]
    \item Compute SVD of $G_{\mathrm{hist}} = U_{\mathrm{hist}}\Sigma V^H$.
          Select top-$P$ pairs $(l_1,l_2)$ by $\sigma_{l_1}\sigma_{l_2}$ (see below).
          Set $A_{\cdot p} = (U_{\mathrm{hist}})_{\cdot l_1(p)}$,
          $B_{\cdot p} = \overline{(U_{\mathrm{hist}})_{\cdot l_2(p)}}$,
          $\Lambda_{kp} = \sigma_{l_1(p)}\sigma_{l_2(p)}V^*_{k,l_1(p)}V_{k,l_2(p)}$.
    \item Orthogonalise $A$ and $B$ via QR; recompute $\boldsymbol{\Lambda} = \mathbf{F}\odot\mathbf{C}$.
    \item Compute $\mathbf{F} = G^T\overline{A}$ and $\mathbf{C} = \overline{G}^T\overline{B}$.
    \item Update $\boldsymbol{\Lambda} = \mathbf{F}\odot\mathbf{C}$.
    \item Update $A_{\cdot p} \leftarrow G(\overline{\Lambda_{\cdot p}}\odot\mathbf{C}_{\cdot p})$ for each $p$; orthogonalise $A$.
    \item Update $B_{\cdot p} \leftarrow \bar{G}(\overline{\Lambda_{\cdot p}}\odot\mathbf{F}_{\cdot p})$ for each $p$; orthogonalise $B$.
    \item Repeat steps 3--6 until $\mathcal{L}$ converges.
\end{enumerate}

\subsubsection*{Optimal SVD initialisation}

The full $L^2$ expansion of the target reads
\begin{equation}
    g_{jk}g^*_{ik}
    = \sum_{l_1=1}^L\sum_{l_2=1}^L
      \sigma_{l_1}\sigma_{l_2}\,U_{jl_1}U^*_{il_2}\,V^*_{kl_1}V_{kl_2}.
    \label{eq:full_expansion}
\end{equation}
Each pair $(l_1, l_2)$ contributes a term of magnitude $\sigma_{l_1}\sigma_{l_2}$.
In our convention $B_{ip} = U^*_{il_2(p)}$ (absorbing the conjugate into $B$),
the pair-$p$ contribution is $\Lambda_{kp}A_{jp}B_{ip} = \sigma_{l_1}\sigma_{l_2}U_{jl_1}U^*_{il_2}V^*_{kl_1}V_{kl_2}$.
Retaining only diagonal pairs $(l,l)$ recovers the hermitian case
$B_{ip}=\Theta^*_{ip}=U^*_{ip}$, $\Lambda_{kp}=\sigma_p^2|V_{kp}|^2=|\Omega_{kp}|^2$.

For the non-hermitian initialisation, rank all $L^2$ pairs by $\sigma_{l_1}\sigma_{l_2}$
and retain the top $P$.
The off-diagonal pair $(1,2)$ contributes $\sigma_1\sigma_2$, which exceeds the diagonal
contribution $\sigma_P^2$ of the last hermitian term whenever
$\sigma_P < \sqrt{\sigma_1\sigma_2}$ — almost always true for a fast-decaying spectrum.
Since repeated indices occur (e.g.\ $U_{\cdot 1}$ appears in pairs $(1,1),(1,2),(1,3),\ldots$),
the $A$ and $B$ columns are not orthonormal as initialised; QR orthogonalisation followed
by a $\boldsymbol{\Lambda}$ recompute (step~2 of the algorithm) restores the constraint.

\section{Gradient Nonlinearity via Coordinate Substitution (Warp)}

\subsection{Separability assumption}

Split the higher-order field terms in $\Delta B(\mathbf{r},t)$ into two classes:
\begin{itemize}
    \item \textbf{Static GNL terms}: spatial harmonics $N^m_l(\mathbf{r})$ ($l\ge2$) whose
          time waveform is, to good approximation, the \emph{same} impulse response as the
          linear term of the physical gradient coil that produced them,
          $G^{lm}_\beta(t) \approx c^{lm}_\beta\, G^0_\beta(t)$. These are pure coil-geometry
          imperfections — fixed spatial pattern, scaled by the same coil current as the
          linear gradient.
    \item \textbf{Residual terms}: $\Delta B_0(\mathbf{r})$ (off-resonance) and concomitant
          fields, whose time dependence is quadratic in the gradient waveforms (e.g.
          $G_x(t)G_z(t)$) and therefore cannot share a single coil's linear waveform.
\end{itemize}
Define, per axis $\beta$, the composite coordinate field
\begin{equation}
    u_\beta(\mathbf{r}) = r_\beta + \sum_{\substack{l\ge2 \\ \text{static}}} c^{lm}_\beta N^m_l(\mathbf{r}),
    \qquad \mathbf{u}(\mathbf{r}) = (u_x,u_y,u_z).
\end{equation}
Under the separability assumption,
\begin{equation}
    \sum_\beta G^0_\beta(t) r_\beta + \sum_{\text{static GNL}} G^{lm}_\beta(t)N^m_l(\mathbf{r})
    = \sum_\beta G^0_\beta(t)\,u_\beta(\mathbf{r}) = \mathbf{k}(t)\cdot\mathbf{u}(\mathbf{r}).
\end{equation}
What remains in $\Phi_{\mathrm{res}}(\mathbf{r},t)$ is only $T_2$ decay, $\Delta B_0$, and
concomitant fields — i.e. $\Phi$ from \eqref{eq:approx1} with the static-GNL channels removed
from $Q$.

\subsection{Coordinate substitution}

With $\mathbf{q}=\mathbf{u}(\mathbf{r})$,
\begin{align}
    S_c(t) &= \int_V s_c(\mathbf{r})\rho(\mathbf{r}) e^{\Phi_{\mathrm{res}}(\mathbf{r},t)}
              e^{-i\mathbf{k}(t)\cdot\mathbf{u}(\mathbf{r})}\, d\mathbf{r} \nonumber\\
           &= \int_{\mathbf{u}(V)} \underbrace{s_c(\mathbf{r}(\mathbf{q}))\rho(\mathbf{r}(\mathbf{q}))
              \,|\det \mathbf{J}_{\mathbf{r}}(\mathbf{q})|}_{\tilde\rho_c(\mathbf{q})}
              \, e^{\Phi_{\mathrm{res}}(\mathbf{r}(\mathbf{q}),t)} e^{-i\mathbf{k}(t)\cdot\mathbf{q}}
              \, d\mathbf{q},
\end{align}
where $\mathbf{J}_{\mathbf{r}} = \mathbf{J}_{\mathbf{u}}^{-1}$. This is exactly the
$\mathbf{A}^H\mathbf{A}$ Toeplitz formalism of the previous section, evaluated in
$\mathbf{q}$-coordinates, with $\Phi_{\mathrm{res}}$ now spanning only off-resonance and
concomitant fields (fewer histogram channels $Q$, no static-GNL blow-up of $n_{\mathrm{hist}}$).

\subsection{Two-grid discretisation}

\begin{itemize}
    \item \textbf{$\mathbf{q}$-grid}: regular Cartesian, chosen at FFT-optimal size — the
          grid the Toeplitz kernel \eqref{eq:toeplitz_apply} is padded/transformed on,
          unchanged from the existing implementation.
    \item \textbf{$\mathbf{r}$-grid}: physical voxel grid (image domain, where the
          regularizer lives), \emph{oversampled} relative to the nominal resolution so that
          $\mathbf{u}(\text{$\mathbf{r}$-grid domain}) \supseteq \text{$\mathbf{q}$-grid
          domain}$ — i.e. no extrapolation in $\mathbf{W}$ below. The $\mathbf{r}$-grid is
          never FFT'd, so its size is unconstrained by factorization concerns.
    \item $\mathbf{W}$: sparse linear interpolation $\mathbf{r}\text{-grid}
          \to \mathbf{q}\text{-grid}$ representing $\mathbf{q}=\mathbf{u}(\mathbf{r})$,
          precomputed once (static, determined by gradient-coil hardware). $\mathbf{W}^H$ is
          its adjoint, $\mathbf{q}\text{-grid}\to\mathbf{r}\text{-grid}$.
\end{itemize}

\subsection{Jacobian weighting folded into coil maps}

$\sqrt{|\det\mathbf{J}_{\mathbf{u}}(\mathbf{r})|}$ is a diagonal, closed-form
(polynomial-in-$\mathbf{r}$, since $\mathbf{u}$ is polynomial) scaling, precomputed once and
absorbed into effective coil sensitivities,
\begin{equation}
    s'_c(\mathbf{r}) = s_c(\mathbf{r})\,\sqrt{|\det\mathbf{J}_{\mathbf{u}}(\mathbf{r})|}.
\end{equation}
This keeps $\mathbf{W}$ close to a partial isometry ($\mathbf{W}^H\mathbf{W}\approx
\mathbf{I}$ on the overlap region), preserving adjoint consistency and CG conditioning
without extra operator applications.

\subsection{Diffeomorphism requirement}

$\mathbf{u}$ must be invertible ($\det\mathbf{J}_{\mathbf{u}}\neq0$) over the oversampled
$\mathbf{r}$-grid domain. $\mathbf{J}_{\mathbf{u}} = \mathbf{I} + (\text{small polynomial
perturbation})$, whose magnitude grows with $|\mathbf{r}|$ (the quadratic GNL terms scale as
$\alpha\,|\mathbf{r}|$ in the Jacobian). This is \emph{not} assumed a priori — verify
$\det\mathbf{J}_{\mathbf{u}}(\mathbf{r})>0$ numerically on the $\mathbf{r}$-grid (cheap,
closed form). If it fails near the FOV edge for a given coefficient set, either crop the
$\mathbf{r}$-grid domain or leave the offending high-order term in $\Phi_{\mathrm{res}}$
instead of absorbing it into $\mathbf{u}$.

\subsection{Resulting operator chain}

\begin{align}
    \mathbf{A}_c &= \mathbf{F}_{\mathbf{q}}\,\mathbf{D}_{\mathrm{res}}\,\mathbf{W}\,
                    \mathrm{diag}(s'_c), \\
    \mathbf{A}_c^H\mathbf{A}_c &= \mathrm{diag}(s'^*_c)\,\mathbf{W}^H\,
                    \mathbf{D}_{\mathrm{res}}^H\,(\text{Toeplitz}_{\mathbf{q}})\,
                    \mathbf{D}_{\mathrm{res}}\,\mathbf{W}\,\mathrm{diag}(s'_c),
\end{align}
where $\text{Toeplitz}_{\mathbf{q}}$ is \eqref{eq:normal_op_PR} unchanged — $\mathbf{k}(t)\cdot\mathbf{q}$
is linear and translation-invariant on the regular $\mathbf{q}$-grid, so the PSF/Toeplitz
embedding is unaffected by the warp. $\mathbf{D}_{\mathrm{res}}$ is the existing low-rank
histogram/SVD interpolator, now over off-resonance + concomitant fields only. $\mathbf{W}$,
$\mathbf{W}^H$ add one sparse interpolation each per normal-operator application —
$O(N)$, negligible next to the Toeplitz FFTs. Image-domain regularizers are applied in
$\mathbf{r}$-space each iteration (i.e. $\mathbf{W}$ sits \emph{inside} the iterative loop,
not just as a final resampling step).
